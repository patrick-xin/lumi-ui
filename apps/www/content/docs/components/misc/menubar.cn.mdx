## 组件结构

<Tabs defaultValue="common">
  <TabsList>
    <TabsTab value="common">常用组合</TabsTab>
    <TabsTab value="primitive">基础组件</TabsTab>
  </TabsList>

  <TabsPanel value="common">
     ```tsx
  <Menubar>
      <MenubarMenu>
        <MenubarMenuTrigger />
        <MenubarMenuContent>
          <MenubarMenuItem />
          <MenubarMenuItem>
            <MenubarMenuShortcut />
          </MenubarMenuItem>
          <MenubarMenuSeparator />
          <MenubarMenuGroup>
            <MenubarMenuGroupLabel />
          </MenubarMenuGroup>
          <MenubarMenuRadioGroup>
            <MenubarMenuRadioItemContent />
          </MenubarMenuRadioGroup>
          <MenubarMenuCheckboxItemContent />
          <MenubarMenuSubMenu>
            <MenubarMenuSubMenuTriggerGroup />
            <MenubarMenuSubMenuContent>
              {/* 子菜单项 */}
            </MenubarMenuSubMenuContent>
          </MenubarMenuSubMenu>
        </MenubarMenuContent>
      </MenubarMenu>
  </Menubar>

     ```
  </TabsPanel>

  <TabsPanel value="primitive">
    ```tsx
    <Menubar>
      <MenubarMenuTrigger />
      <MenubarMenuPortal>
        <MenubarMenuBackdrop />
        <MenubarMenuPositioner>
          <MenubarMenuPopup>
            <MenubarMenuArrow />
            <MenubarMenuItem />
            <MenubarMenuSeparator />
            <MenubarMenuGroup>
              <MenubarMenuGroupLabel />
            </MenubarMenuGroup>
            <MenubarMenuRadioGroup>
              <MenubarMenuRadioItem />
            </MenubarMenuRadioGroup>
            <MenubarMenuCheckboxItem />
            <MenubarMenuSubmenuRoot>
              <MenubarMenuSubmenuTrigger />
            </MenubarMenuSubmenuRoot>
          </MenubarMenuPopup>
        </MenubarMenuPositioner>
      </MenubarMenuPortal>
    </Menubar>

```
  </TabsPanel>
</Tabs>
