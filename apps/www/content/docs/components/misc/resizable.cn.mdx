## 组件结构
```tsx
  <ResizableGroup>   
    <ResizablePanel />
    <ResizableSeparator />     
  </ResizableGroup>
```

## 基本用法

```tsx
import {
  ResizableGroup,
  ResizablePanel,
} from "@components/ui/resizable";

<ResizableGroup orientation="horizontal">
  <ResizablePanel>面板一</ResizablePanel>
  <ResizablePanel>面板二</ResizablePanel>
</ResizableGroup>
```
