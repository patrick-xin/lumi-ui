## 组件结构

```tsx
 <Avatar>
  <AvatarImage/>
  <AvatarFallback/>
 </Avatar>
```

## 基本用法

```tsx
import { Avatar, AvatarFallback, AvatarImage } from "@/components/ui/avatar"
```

```tsx
<Avatar>
  <AvatarImage src="https://github.com/patrick-xin.png" />
  <AvatarFallback>CN</AvatarFallback>
</Avatar>
```
