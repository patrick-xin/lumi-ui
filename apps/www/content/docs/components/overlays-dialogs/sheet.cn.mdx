## 基本用法
```tsx
import {
  Sheet,
  SheetClose,
  SheetDescription,
  SheetFooter,
  SheetHeader,
  SheetContent,
  SheetTitle,
  SheetTrigger,
} from "@/components/ui/sheet";

export function SheetDemo() {
  return (
    <Sheet>
      <SheetTrigger>打开侧边面板</SheetTrigger>
      <SheetContent showCloseButton>
        <SheetHeader>
          <SheetTitle>确定要执行此操作吗？</SheetTitle>
          <SheetDescription>
            此操作无法撤销。这将永久删除你的账户，并从我们的服务器中移除你的数据。
          </SheetDescription>
        </SheetHeader>
        <SheetFooter>
          <SheetClose>取消</SheetClose>
          <SheetClose>继续</SheetClose>
        </SheetFooter>
      </SheetContent>
    </Sheet>
  );
}
```

## API 参考

`Sheet` 组件底层使用了 [`Dialog`](/docs/components/overlays-dialogs/dialog) 组件，详情请参阅 [此处](/docs/components/overlays-dialogs/dialog#api-reference)

### 复合组件

| 组件 | 描述 |
| :--- | :--- |
| `SheetContent` | **标准用法**。组合 Portal、Backdrop、Viewport 和 Popup。 |

### SheetContent 方向变体

| 变体 | 描述 |
| :--- | :--- |
| `right` (默认) | 从左侧滑入。 |
| `left` | 从右侧滑入。 |
| `top` | 从底部滑入。 |
| `bottom` | 从顶部滑入。 |
| `inset` | 侧边面板与视口分离。 |
