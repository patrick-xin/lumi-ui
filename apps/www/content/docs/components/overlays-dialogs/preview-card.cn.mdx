## 基本用法

```tsx
import {
  PreviewCard,
  PreviewCardTrigger,
  PreviewCardContent,
} from "@/components/ui/preview-card"
```

```tsx
<PreviewCard>
  <PreviewCardTrigger>悬停查看</PreviewCardTrigger>
  <PreviewCardContent>
    The React Framework – created and maintained by @vercel.
  </PreviewCardContent>
</PreviewCard>
```
