## 基本用法

```tsx
import {
  Tooltip,
  TooltipTrigger,
  TooltipContent,
} from "@/components/ui/tooltip";
```

```tsx
<Tooltip>
  <TooltipTrigger>悬停查看</TooltipTrigger>
  <TooltipContent>
    工具提示
  </TooltipContent>
</Tooltip>
```
